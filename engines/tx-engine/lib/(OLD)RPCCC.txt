const { Connection, PublicKey } = require('@solana/web3.js');
function createRotatingFetch(urls) {
  let index = 0;
  return async function customFetch(url, options) {
    let lastError;
    const maxAttempts = urls.length * 2;
    for (let attempt = 0; attempt < maxAttempts; attempt++) {
      const currentRpcUrl = urls[index];
      index = (index + 1) % urls.length;
      try {
        const response = await fetch(currentRpcUrl, options);
        if (response.status === 429) {
          lastError = new Error(`[RPC ${index}] 429 Too Many Requests`);
          await new Promise(res => setTimeout(res, 500));
          continue;
        }
        if (!response.ok) {
          lastError = new Error(`[RPC ${index}] HTTP Error: ${response.status}`);
          continue;
        }
        return response;
      } catch (error) {
        lastError = error;
      }
    }
    throw new Error(`All RPC fetch attempts failed. Last error: ${lastError.message}`);
  };
}
function createConnection(rpcUrl, commitment = 'confirmed') {
  const urlsArray = [rpcUrl];
  return new Connection(rpcUrl, {
    commitment,
    confirmTransactionInitialTimeout: 60_000,
    fetch: createRotatingFetch(urlsArray)
  });
}
async function fetchMintInfo(connection, mintAddress, commitment) {
  const mint = new PublicKey(mintAddress);
  const mintInfo = await connection.getParsedAccountInfo(mint, commitment);
  const owner = mintInfo.value?.owner?.toBase58?.() ?? null;
  return { mintInfo, owner };
}
async function fetchTokenAccountsByMint(connection, mintAddress, commitment, tokenProgramId) {
  const mint = new PublicKey(mintAddress);
  const programId = new PublicKey(tokenProgramId);
  return connection.getParsedProgramAccounts(programId, {
    commitment,
    filters: [
      {
        memcmp: {
          offset: 0,
          bytes: mint.toBase58()
        }
      }
    ]
  });
}
module.exports = {
  createConnection,
  fetchMintInfo,
  fetchTokenAccountsByMint
};
